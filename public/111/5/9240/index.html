Eh oui, il fallait bien que ça arrive. Voilà donc, j'ai des suggestions pour de nouvelles potions, et que je ne voie aucun d'entre vous se plaindre, alchimie est la seule classe artisanne qui n'a jamais été retouchée.

1. Filtre de mana regénération
**2. ** Filtre d'amour (potions rp)
3. Filtre de métamorphose
**4. ** Drogue hallucinogène
**5. ** Sébum de vérité (potions rp)

Donc le 1er : Mana Regénération

Je propose qu'il y ait 3 forces comme les potions de base.
-*Lesser mana regen * (1-10 mana)
-*Mana regen * (5-20 mana)
-Greater Mana regen (15-30 mana)

Les potions devront se faire avec des ailes de chauve souris.

Le 2e : Philtre d'Amour (potion rp)

Donc comme je le mentionne, le philtre d'amour serait avant tout une potion au niveau rp. Il serait utilisable avec le skill Empoisonnement selon le pourcentage.

Donc le philtre d'amour ne pourrait être mis que sur de la nourriture (difficile d'aimer une personne qui te plante une dague dans la peau) et lorsqu'ingéré par la victime, une petite boîte de choix apparaîtrait:

Vous avez ingéré un filtre d'amour. Deux cases y seront pour le choix. Un "X" pour non, je n'accepte pas de jouer les effets, donc le filtre n'a pas eu d'effet. Un "Y" pour oui, j'accepte de jouer le jeu pour un délais qui sera prédéterminé par l'équipe, moi je dirais 1 heure irl. Donc il y aurait un "timer" pour délimiter les effets, comme quoi ce n'est pas irréversible, et les personnes qui désirent l'utiliser devront "enchanter" leur victime régulièrement s'ils veulent que le délais continue.

Le 3e : Le Philtre de Métamorphose

Il permettrait de pouvoir se transformer pour une période de temps en une créature particulière, ou bien aléatoire. Je n'ai pas vraiment élaboré sur cette idée et donc le thread est ouvert pour les débats à son sujet.

Le 4e : Drogue Hallucinogène

La drogue hallucinogène fonctionnerait un peu comme le sort Hallucination de base sous sphère pour une période de 45 secondes. Mais il serait également applicable avec le skill "Empoisonnement" selon un pourcentage donné.

Le 5e : Sébum de vérité (potion rp)

Le sébum de vérité, serait, comme mentionné une potion rp. Donc il fonctionnerait à la base exactement comme le Philtre d'amour.

J'aimerais des commentaires constructifs s'il vous plait ainsi que des améliorations ou bien d'autres idées de potions pour pouvoir améliorer la classe alchimiste afin que d'autres s'y intéressent.

FICHIER: BasePolymorphPotion.cs
using System;
using Server;

namespace Server.Items
{
\tpublic abstract class BasePolymorphPotion : BasePotion
\t{
\t\tpublic abstract TimeSpan Duration{ get; }

\t\tpublic BasePolymorphPotion( PotionEffect effect ) : base( 0xE26, effect )
\t\t{
\t\t}

\t\tpublic BasePolymorphPotion( Serial serial ) : base( serial )
\t\t{
\t\t}

\t\tpublic override void Serialize( GenericWriter writer )
\t\t{
\t\t\tbase.Serialize( writer );

\t\t\twriter.Write( (int) 0 ); // version
\t\t}

\t\tpublic override void Deserialize( GenericReader reader )
\t\t{
\t\t\tbase.Deserialize( reader );

\t\t\tint version = reader.ReadInt();
\t\t}

\t\tpublic bool DoPolymorph( Mobile from )
\t\t{
\t\t\tif ( from.BodyValue == 401 )
\t\t\t{
\t\t\tfrom.BodyValue = 400;
\t\t\t}
\t\t\telse if ( from.BodyValue == 400 )
\t\t\t{
\t\t\tfrom.BodyValue = 401;\t
\t\t\t}
\t\t\treturn true;
\t\t}

\t\tpublic override void Drink( Mobile from )
\t\t{
\t\t\tif ( DoPolymorph( from ) )
\t\t\t{
\t\t\t\tBasePotion.PlayDrinkEffect( from );

\t\t\t\tthis.Consume();
\t\t\t}
\t\t}
\t}
}

using System;
using Server.Items;

namespace Server.Engines.Craft
{
\tpublic class DefAlchemy : CraftSystem
\t{
\t\tpublic override SkillName MainSkill
\t\t{
\t\t\tget\t{ return SkillName.Alchemy;\t}
\t\t}

\t\tpublic override int GumpTitleNumber
\t\t{
\t\t\tget { return 1044001; } //

ALCHEMY MENU

\t\t}

\t\tprivate static CraftSystem m_CraftSystem;

\t\tpublic static CraftSystem CraftSystem
\t\t{
\t\t\tget
\t\t\t{
\t\t\t\tif ( m_CraftSystem == null )
\t\t\t\t\tm_CraftSystem = new DefAlchemy();

\t\t\t\treturn m_CraftSystem;
\t\t\t}
\t\t}

\t\tpublic override double GetChanceAtMin( CraftItem item )
\t\t{
\t\t\treturn 0.0; // 0%
\t\t}

\t\tprivate DefAlchemy() : base( 1, 1, 1.25 )// base( 1, 1, 3.1 )
\t\t{
\t\t}

\t\tpublic override int CanCraft( Mobile from, BaseTool tool, Type itemType )
\t\t{
\t\t\tif( tool == null || tool.Deleted || tool.UsesRemaining < 0 )
\t\t\t\treturn 1044038; // You have worn out your tool!
\t\t\telse if ( !BaseTool.CheckAccessible( tool, from ) )
\t\t\t\treturn 1044263; // The tool must be on your person to use.

\t\t\treturn 0;
\t\t}

\t\tpublic override void PlayCraftEffect( Mobile from )
\t\t{
\t\t\tfrom.PlaySound( 0x242 );
\t\t}

\t\tprivate static Type typeofPotion = typeof( BasePotion );

\t\tpublic static bool IsPotion( Type type )
\t\t{
\t\t\treturn typeofPotion.IsAssignableFrom( type );
\t\t}

\t\tpublic override int PlayEndingEffect( Mobile from, bool failed, bool lostMaterial, bool toolBroken, int quality, bool makersMark, CraftItem item )
\t\t{
\t\t\tif ( toolBroken )
\t\t\t\tfrom.SendLocalizedMessage( 1044038 ); // You have worn out your tool

\t\t\tif ( failed )
\t\t\t{
\t\t\t\tif ( IsPotion( item.ItemType ) )
\t\t\t\t{
\t\t\t\t\tfrom.AddToBackpack( new Bottle() );
\t\t\t\t\treturn 500287; // You fail to create a useful potion.
\t\t\t\t}
\t\t\t\telse
\t\t\t\t{
\t\t\t\t\treturn 1044043; // You failed to create the item, and some of your materials are lost.
\t\t\t\t}
\t\t\t}
\t\t\telse
\t\t\t{
\t\t\t\tfrom.PlaySound( 0x240 ); // Sound of a filling bottle

\t\t\t\tif ( IsPotion( item.ItemType ) )
\t\t\t\t{
\t\t\t\t\tif ( quality == -1 )
\t\t\t\t\t\treturn 1048136; // You create the potion and pour it into a keg.
\t\t\t\t\telse
\t\t\t\t\t\treturn 500279; // You pour the potion into a bottle...
\t\t\t\t}
\t\t\t\telse
\t\t\t\t{
\t\t\t\t\treturn 1044154; // You create the item.
\t\t\t\t}
\t\t\t}
\t\t}
