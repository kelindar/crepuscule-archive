Post by Feydanith, dit la Carpe - April 30, 2007 at 5:57 PM

J'ai fait ça sur le pouce :

Ca copie fringues et titre.

using System;
using System.Reflection;
using Server.Items;
using Server.Targeting;

namespace Server.Scripts.Commands
{
\tpublic class getFringues
\t{
\t\tpublic static void Initialize()
\t\t{
\t\t\tServer.Commands.Register( "getFringues", AccessLevel.GameMaster, new CommandEventHandler( getFringues_OnCommand ) );
\t\t}

\t\t[Usage( "getFringues" )]
\t\t[Description( "Attribue les fringues d'un mobile à un autre." )]
\t\tprivate static void getFringues_OnCommand( CommandEventArgs e )
\t\t{
\t\t\te.Mobile.Target = new getFringuesTargetFrom();
\t\t\te.Mobile.SendMessage( "De quel mobile voulez-vous copier les fringues ?" );
\t\t}

\t\tprivate class getFringuesTargetFrom : Target
\t\t{
\t\t\tpublic getFringuesTargetFrom() : base( -1, true, TargetFlags.None )
\t\t\t{
\t\t\t}

\t\t\tprotected override void OnTarget( Mobile from, object o )
\t\t\t{
\t\t\t\tif ( o is Mobile )
\t\t\t\t{
\t\t\t\t\tfrom.Target = new getFringuesTargetTo( (Mobile)o );
\t\t\t\t\tfrom.SendMessage( "Sur quel mobile voulez-vous appliquer ces fringues ?" );
\t\t\t\t}
\t\t\t\telse
\t\t\t\t\tfrom.SendMessage( "La cible n'est pas un mobile." );
\t\t\t}

\t\t\tprivate class getFringuesTargetTo : Target
\t\t\t{
\t\t\t\tMobile TargetFrom;
\t\t\t\t
\t\t\t\tprivate Layer[] m_layers = new Layer[]
\t\t\t\t{
\t\t\t\t\tLayer.FirstValid,
\t\t\t\t\tLayer.TwoHanded,
\t\t\t\t\tLayer.Shoes,
\t\t\t\t\tLayer.Pants,
\t\t\t\t\tLayer.Shirt,
\t\t\t\t\tLayer.Helm,
\t\t\t\t\tLayer.Gloves,
\t\t\t\t\tLayer.Ring,
\t\t\t\t\tLayer.Unused_x9,
\t\t\t\t\tLayer.Neck,
\t\t\t\t\tLayer.Hair,
\t\t\t\t\tLayer.Waist,
\t\t\t\t\tLayer.InnerTorso,
\t\t\t\t\tLayer.Bracelet,
\t\t\t\t\tLayer.Unused_xF,
\t\t\t\t\tLayer.FacialHair,
\t\t\t\t\tLayer.MiddleTorso,
\t\t\t\t\tLayer.Earrings,
\t\t\t\t\tLayer.Arms,
\t\t\t\t\tLayer.Cloak,
\t\t\t\t\tLayer.OuterTorso,
\t\t\t\t\tLayer.OuterLegs,
\t\t\t\t\tLayer.LastUserValid
\t\t\t\t};
\t\t\t\t
\t\t\t\tpublic getFringuesTargetTo( Mobile m_TargetFrom ) : base( -1, true, TargetFlags.None )
\t\t\t\t{\t\t
\t\t\t\t\tTargetFrom = m_TargetFrom;
\t\t\t\t}

\t\t\t\tprotected override void OnTarget( Mobile from, object o )
\t\t\t\t{
\t\t\t\t\tif ( o is Mobile )
\t\t\t\t\t{
\t\t\t\t\t\tMobile TargetTo = (Mobile)o;
\t\t\t\t\t\tcopyFringues( TargetFrom, TargetTo );
\t\t\t\t\t}
\t\t\t\t\telse
\t\t\t\t\t\tfrom.SendMessage( "La cible n'est pas un mobile." );
\t\t\t\t}

\t\t\t\tpublic void copyFringues( Mobile TargetFrom, Mobile TargetTo )
\t\t\t\t{
\t\t\t\t\t
\t\t\t\t\tTargetTo.Title = TargetFrom.Title;
\t\t\t\t\tUnloadFromAllLayers( TargetTo );
\t\t\t\t\tCopyFromAllLayers( TargetFrom, TargetTo );
\t\t\t\t}

\t\t\t\tprivate void CopyFromAllLayers ( Mobile TargetFrom, Mobile TargetTo )
\t\t\t\t{
\t\t\t\t\tfor ( int i = 0; i < m_layers.Length; ++i )
\t\t\t\t\t\tCopyFromLayer( TargetFrom, TargetTo, m_layers[i] );
\t\t\t\t}
\t
\t\t\t\tprivate void CopyFromLayer( Mobile TargetFrom, Mobile TargetTo, Layer layer )
\t\t\t\t{
\t\t\t\t\tif ( TargetFrom.FindItemOnLayer( layer ) != null )
\t\t\t\t\t{
\t\t\t\t\t\tItem ToCopy = (Item) TargetFrom.FindItemOnLayer( layer );
\t\t\t\t\t\tItem Copy = ToCopy.Dupe( 1 );
\t\t\t\t\t\tTargetTo.EquipItem( Copy );
\t\t\t\t\t}
\t\t\t\t}
\t\t
\t\t\t\tprivate void UnloadFromAllLayers( Mobile TargetTo )
\t\t\t\t{
\t\t\t\t\tfor ( int i = 0; i < m_layers.Length; ++i )
\t\t\t\t\t\tUnloadFromLayer( TargetTo, m_layers[i] );
\t\t\t\t}
\t\t
\t\t\t\tprivate void UnloadFromLayer( Mobile TargetTo, Layer layer )
\t\t\t\t{
\t\t\t\t\tif ( TargetTo.FindItemOnLayer( layer ) != null )
\t\t\t\t\tTargetTo.AddToBackpack( TargetTo.FindItemOnLayer( layer ) );
\t\t\t\t}
\t\t\t}
\t\t}
\t}
}
