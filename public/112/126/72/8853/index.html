scripts/items/misc/interiordecorator.cs

4 points cardinaux ajoutés pour les déplacements.
Refuse de déplacer les objets trop lourds.
Refuse de déplacer les objets trop loins.
Refuse de déplacer les objets lock gm.
Accepte de déplacer le reste !

J'ai pas testé à fond le dernier point, n'ayant poas tout sur ma machine.

using System;
using Server;
using Server.Network;
using Server.Regions;
using Server.Multis;
using Server.Gumps;
using Server.Targeting;

namespace Server.Items
{
\tpublic enum DecorateCommand
\t{
\t\tNone,
\t\tTurn,
\t\tUp,
\t\tDown,
\t\tLeft,
\t\tRight,
\t\tFront,
\t\tBack
\t}

\tpublic class InteriorDecorator : Item
\t{
\t\tprivate DecorateCommand m_Command;

\t\t[CommandProperty( AccessLevel.GameMaster )]
\t\tpublic DecorateCommand Command{ get{ return m_Command; } set{ m_Command = value; InvalidateProperties(); } }

\t\t[Constructable]
\t\tpublic InteriorDecorator() : base( 0xFC1 )
\t\t{
\t\t\tWeight = 1.0;
\t\t\tLootType = LootType.Blessed;
\t\t}

\t\tpublic override int LabelNumber{ get{ return 1041280; } } // an interior decorator

\t\tpublic InteriorDecorator( Serial serial ) : base( serial )
\t\t{
\t\t}

\t\tpublic override void GetProperties( ObjectPropertyList list )
\t\t{
\t\t\tbase.GetProperties( list );
\t\t\t
\t\t\tString text = "";

\t\t\tswitch ( m_Command )
\t\t\t{
\t\t\t\tcase DecorateCommand.Up: \ttext = "Haut"; \t\tbreak;
\t\t\t\tcase DecorateCommand.Down:\ttext = "Bas";\t\tbreak;
\t\t\t\tcase DecorateCommand.Turn:\ttext = "Tourner"; \tbreak;
\t\t\t\tcase DecorateCommand.Left:\ttext = "Sud";\t\tbreak;
\t\t\t\tcase DecorateCommand.Right:\ttext = "Nord";\t\tbreak;
\t\t\t\tcase DecorateCommand.Front:\ttext = "Est";\t\tbreak;
\t\t\t\tcase DecorateCommand.Back:\ttext = "Ouest";\t\tbreak;
\t\t\t}

\t\t\tif ( text != "" )
\t\t\t\tlist.Add( text ); // Turn/Up/Down/Left/Right/Front/Back
\t\t}

\t\tpublic override void Serialize( GenericWriter writer )
\t\t{
\t\t\tbase.Serialize( writer );

\t\t\twriter.Write( (int) 0 ); // version
\t\t}

\t\tpublic override void Deserialize( GenericReader reader )
\t\t{
\t\t\tbase.Deserialize( reader );

\t\t\tint version = reader.ReadInt();
\t\t}

\t\tpublic override void OnDoubleClick( Mobile from )
\t\t{
\t\t\tif ( !CheckUse( this, from ) )
\t\t\t\treturn;

\t\t\tif ( m_Command == DecorateCommand.None )
\t\t\t\tfrom.SendGump( new InternalGump( this ) );
\t\t\telse
\t\t\t\tfrom.Target = new InternalTarget( this );
\t\t}

\t\tpublic static bool CheckUse( InteriorDecorator tool, Mobile from )
\t\t{
\t\t\tif ( tool.Deleted || !tool.IsChildOf( from.Backpack ) )
\t\t\t\tfrom.SendLocalizedMessage( 1042001 ); // That must be in your pack for you to use it.
\t\t\telse
\t\t\t\treturn true;

\t\t\treturn false;
\t\t}

\t\tprivate class InternalGump : Gump
\t\t{
\t\t\tprivate InteriorDecorator m_Decorator;

\t\t\tpublic InternalGump( InteriorDecorator decorator ) : base( 350, 50 )
\t\t\t{
\t\t\t\tm_Decorator = decorator;

\t\t\t\tAddBackground( 0, 0, 200, 400, 2600 );

\t\t\t\tAddButton( 50, 45, 2152, 2154, 1, GumpButtonType.Reply, 0 );
\t\t\t\tAddHtmlLocalized( 90, 50, 70, 40, 1018323, false, false ); // Turn

\t\t\t\tAddButton( 50, 95, 2152, 2154, 2, GumpButtonType.Reply, 0 );
\t\t\t\tAddHtmlLocalized( 90, 100, 70, 40, 1018324, false, false ); // Up

\t\t\t\tAddButton( 50, 145, 2152, 2154, 3, GumpButtonType.Reply, 0 );
\t\t\t\tAddHtmlLocalized( 90, 150, 70, 40, 1018325, false, false ); // Down

\t\t\t\tAddButton( 50, 195, 2152, 2154, 4, GumpButtonType.Reply, 0 );
\t\t\t\tAddHtml( 90, 200, 70, 40, String.Format("Sud"), false, false ); //South

\t\t\t\tAddButton( 50, 245, 2152, 2154, 5, GumpButtonType.Reply, 0 );
\t\t\t\tAddHtml( 90, 250, 70, 40, String.Format("Nord"), false, false ); //North

\t\t\t\tAddButton( 50, 295, 2152, 2154, 6, GumpButtonType.Reply, 0 );
\t\t\t\tAddHtml( 90, 300, 70, 40, String.Format("Est"), false, false ); //East

\t\t\t\tAddButton( 50, 345, 2152, 2154, 7, GumpButtonType.Reply, 0 );
\t\t\t\tAddHtml( 90, 350, 70, 40, String.Format("Ouest"), false, false ); //West

\t\t\t}

\t\t\tpublic override void OnResponse( NetState sender, RelayInfo info )
\t\t\t{
\t\t\t\tDecorateCommand command = DecorateCommand.None;

\t\t\t\tswitch ( info.ButtonID )
\t\t\t\t{
\t\t\t\t\tcase 1: command = DecorateCommand.Turn; break;
\t\t\t\t\tcase 2: command = DecorateCommand.Up; break;
\t\t\t\t\tcase 3: command = DecorateCommand.Down; break;
\t\t\t\t\tcase 4: command = DecorateCommand.Left; break;
\t\t\t\t\tcase 5: command = DecorateCommand.Right; break;
\t\t\t\t\tcase 6: command = DecorateCommand.Front; break;
\t\t\t\t\tcase 7: command = DecorateCommand.Back; break;
\t\t\t\t}

\t\t\t\tif ( command != DecorateCommand.None )
\t\t\t\t{
\t\t\t\t\tm_Decorator.Command = command;
\t\t\t\t\tsender.Mobile.Target = new InternalTarget( m_Decorator );
\t\t\t\t}
\t\t\t}
\t\t}

\t\tprivate class InternalTarget : Target
\t\t{
\t\t\tprivate InteriorDecorator m_Decorator;

\t\t\tpublic InternalTarget( InteriorDecorator decorator ) : base( -1, false, TargetFlags.None )
\t\t\t{
\t\t\t\tCheckLOS = false;

\t\t\t\tm_Decorator = decorator;
\t\t\t}

\t\t\tprotected override void OnTargetNotAccessible( Mobile from, object targeted )
\t\t\t{
\t\t\t\tOnTarget( from, targeted );
\t\t\t}

\t\t\tprotected override void OnTarget( Mobile from, object targeted )
\t\t\t{
\t\t\t\tif ( targeted == m_Decorator )
\t\t\t\t{
\t\t\t\t\tm_Decorator.Command = DecorateCommand.None;
\t\t\t\t\tfrom.SendGump( new InternalGump( m_Decorator ) );
\t\t\t\t}
\t\t\t\telse if ( targeted is Item && InteriorDecorator.CheckUse( m_Decorator, from ) )
\t\t\t\t{
\t\t\t\t\tItem item = (Item)targeted;
\t\t\t\t\t
\t\t\t\t\tif ( item.TotalWeight + item.PileWeight > 100 )
\t\t\t\t\t{
\t\t\t\t\t\tfrom.SendLocalizedMessage( 1042272 ); // That is too heavy.
\t\t\t\t\t}
\t\t\t\t\t
\t\t\t\t\telse if ( !from.InRange( item.GetWorldLocation(), 1 ) )
\t\t\t\t\t{
\t\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas l'atteindre." );
\t\t\t\t\t}
\t\t\t\t\t
\t\t\t\t\telse if ( !item.Movable && !item.IsLockedDown )
\t\t\t\t\t{
\t\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas bouger ceci." );
\t\t\t\t\t}
\t\t\t\t\t
\t\t\t\t\telse
\t\t\t\t\t{
\t\t\t\t\t\tswitch ( m_Decorator.Command )
\t\t\t\t\t\t{
\t\t\t\t\t\t\tcase DecorateCommand.Up:\tUp( item, from );\tbreak;
\t\t\t\t\t\t\tcase DecorateCommand.Down:\tDown( item, from );\tbreak;
\t\t\t\t\t\t\tcase DecorateCommand.Turn:\tTurn( item, from );\tbreak;
\t\t\t\t\t\t\tcase DecorateCommand.Left:\tLeft( item, from );\tbreak;
\t\t\t\t\t\t\tcase DecorateCommand.Right:\tRight( item, from );\tbreak;
\t\t\t\t\t\t\tcase DecorateCommand.Front:\tFront( item, from );\tbreak;
\t\t\t\t\t\t\tcase DecorateCommand.Back:\tBack( item, from );\tbreak;
\t\t\t\t\t\t}
\t\t\t\t\t}
\t\t\t\t}
\t\t\t}

\t\t\tprivate static void Turn( Item item, Mobile from )
\t\t\t{
\t\t\t\tFlipableAttribute[] attributes = (FlipableAttribute[])item.GetType().GetCustomAttributes( typeof( FlipableAttribute ), false );

\t\t\t\tif( attributes.Length > 0 )
\t\t\t\t\tattributes[0].Flip( item );
\t\t\t\telse
\t\t\t\t\tfrom.SendLocalizedMessage( 1042273 ); // You cannot turn that.
\t\t\t}

\t\t\tprivate static void Up( Item item, Mobile from )
\t\t\t{
\t\t\t\tint floorZ = GetFloorZ( item );

\t\t\t\tif ( floorZ > int.MinValue && item.Z < (floorZ + 15) ) // Confirmed : no height checks here
\t\t\t\t\titem.Location = new Point3D( item.Location, item.Z + 1 );
\t\t\t\telse
\t\t\t\t\tfrom.SendLocalizedMessage( 1042274 ); // You cannot raise it up any higher.
\t\t\t}

\t\t\tprivate static void Down( Item item, Mobile from )
\t\t\t{
\t\t\t\tint floorZ = GetFloorZ( item );

\t\t\t\tif ( floorZ > int.MinValue && item.Z > GetFloorZ( item ) )
\t\t\t\t\titem.Location = new Point3D( item.Location, item.Z - 1 );
\t\t\t\telse
\t\t\t\t\tfrom.SendLocalizedMessage( 1042275 ); // You cannot lower it down any further.
\t\t\t}

\t\t\tprivate static void Left( Item item, Mobile from )
\t\t\t{
\t\t\t\titem.Y = item.Y + 1 ;
\t\t\t}

\t\t\tprivate static void Front( Item item, Mobile from )
\t\t\t{
\t\t\t\titem.X = item.X + 1;
\t\t\t}
\t\t\tprivate static void Right( Item item, Mobile from )
\t\t\t{
\t\t\t\titem.Y = item.Y - 1;
\t\t\t}
\t\t\tprivate static void Back( Item item, Mobile from )
\t\t\t{
\t\t\t\titem.X = item.X - 1;
\t\t\t}
\t\t\tprivate static int GetFloorZ( Item item )
\t\t\t{
\t\t\t\tMap map = item.Map;

\t\t\t\tif ( map == null )
\t\t\t\t\treturn int.MinValue;

\t\t\t\tTile[] tiles = map.Tiles.GetStaticTiles( item.X, item.Y, true );

\t\t\t\tint z = int.MinValue;

\t\t\t\tfor ( int i = 0; i < tiles.Length; ++i )
\t\t\t\t{
\t\t\t\t\tTile tile = tiles[i];
\t\t\t\t\tItemData id = TileData.ItemTable[tile.ID & 0x3FFF];

\t\t\t\t\tint top = tile.Z; // Confirmed : no height checks here

\t\t\t\t\tif ( id.Surface && !id.Impassable && top > z && top <= item.Z )
\t\t\t\t\t\tz = top;
\t\t\t\t}

\t\t\t\treturn z;
\t\t\t}
\t\t}
\t}
}
