Dupe amélioré.

Post by Feydanith, dit la Carpe - May 28, 2007 at 5:52 PM

Comme son nom l'indique, un dupe amélioré pour crépuscule.
En gros, c'est un Dupe qui ne dupe que directement dans le sac, mais celui-ci permet de copier un conteneur, avec son contenu et si celui-ci contient d'autres conteneurs, le contenu de ceux-ci récursivement.

.DupeCrep amount

Pas sorcier à utiliser ^^
Il est déjà installé et en jeu.

using System;
using System.Reflection;
using Server.Items;
using Server.Targeting;

namespace Server.Scripts.Commands
{
\tpublic class DupeCrep
\t{
\t\tpublic static void Initialize()
\t\t{
\t\t\tServer.Commands.Register( "DupeCrep", AccessLevel.GameMaster, new CommandEventHandler( DupeCrep_OnCommand ) );
\t\t}

\t\t[Usage( "DupeCrep [amount]" )]
\t\t[Description( "Duplique un conteneur et son contenu ou un objet." )]
\t\tprivate static void DupeCrep_OnCommand( CommandEventArgs e )
\t\t{
\t\t\tint amount = 1;
\t\t\tif ( e.Length >= 1 )
\t\t\t\tamount = e.GetInt32( 0 );
\t\t\te.Mobile.Target = new DupeCrepTarget( amount > 0 ? amount : 1 );
\t\t\te.Mobile.SendMessage( "Quel conteneur ou objet voulez-vous dupliquer ?" );
\t\t}

\t\tprivate class DupeCrepTarget : Target
\t\t{
\t\t\tprivate int m_Amount;

\t\t\tpublic DupeCrepTarget( int amount ) : base( 15, false, TargetFlags.None )
\t\t\t{
\t\t\t\tm_Amount = amount;
\t\t\t}

\t\t\tprotected override void OnTarget( Mobile from, object targ )
\t\t\t{\t\t\t\t
\t\t\t\tContainer pack = from.Backpack;
\t\t\t\t
\t\t\t\tif ( !(targ is Container) && !(targ is Item) )
\t\t\t\t{
\t\t\t\t\tfrom.SendMessage( "Vous pouvez uniquement dupliquer des conteneurs ou des objets." );
\t\t\t\t\treturn;
\t\t\t\t}
\t\t\t\t\t\t\t\t
\t\t\t\telse if ( targ is Container )
\t\t\t\t{
\t\t\t\t\tfor(int i = 0; i < m_Amount; ++i)\t\t\t\t\t
\t\t\t\t\t\tcopyContainer( from, pack, (Container)targ );
\t\t\t\t}
\t\t\t\telse
\t\t\t\t{
\t\t\t\t\tfor(int i = 0; i < m_Amount; ++i)
\t\t\t\t\t\tcopyItem( from, pack, (Item)targ );
\t\t\t\t}
\t\t\t}
\t\t\t
\t\t\tprivate static Item copyItem( Mobile from, Container pack, Item toCopy )
\t\t\t{
\t\t\t\tType t = toCopy.GetType();
\t\t\t\tConstructorInfo[] info = t.GetConstructors();
\t\t\t\t\t\t\t
\t\t\t\tforeach ( ConstructorInfo c in info )
\t\t\t\t{
\t\t\t\t\tParameterInfo[] paramInfo = c.GetParameters();
\t\t\t\t\tif ( paramInfo.Length == 0 )
\t\t\t\t\t{
\t\t\t\t\t\tobject[] objParams = new object[0];
\t\t\t\t\t\ttry
\t\t\t\t\t\t{
\t\t\t\t\t\t\tobject o = c.Invoke( objParams );
\t\t\t\t\t\t\tif ( o != null && o is Item )
\t\t\t\t\t\t\t{
\t\t\t\t\t\t\t\tItem copy = (Item)o;
\t\t\t\t\t\t\t\ttoCopy.Dupe( copy, toCopy.Amount );
\t\t\t\t\t\t\t\tPropertyInfo[] props = toCopy.GetType().GetProperties();
\t\t\t\t\t\t\t\tfor ( int j = 0; j < props.Length; j++ )
\t\t\t\t\t\t\t\t{
\t\t\t\t\t\t\t\t\tif ( props[j].CanRead && props[j].CanWrite )
\t\t\t\t\t\t\t\t\t\tprops[j].SetValue( copy, props[j].GetValue( toCopy, null ), null );
\t\t\t\t\t\t\t\t}
\t\t\t\t\t\t\t\tif ( pack != null )
\t\t\t\t\t\t\t\t\tpack.DropItem( copy );
\t\t\t\t\t\t\t\telse
\t\t\t\t\t\t\t\t\tcopy.MoveToWorld( from.Location, from.Map );
\t\t\t\t\t\t\t\treturn copy;
\t\t\t\t\t\t\t}
\t\t\t\t\t\t}
\t\t\t\t\t\tcatch
\t\t\t\t\t\t{
\t\t\t\t\t\t\tfrom.SendMessage( "Erreur lors de la copie d'un objet !" );
\t\t\t\t\t\t\treturn null;
\t\t\t\t\t\t}
\t\t\t\t\t}
\t\t\t\t}
\t\t\t\treturn null;
\t\t\t}
\t\t\t
\t\t\tprivate static void copyContainer( Mobile from, Container pack, Container toCopy )
\t\t\t{
\t\t\t\tContainer copy = (Container)copyItem( from, pack, toCopy );
\t\t\t\t
\t\t\t\tif ( copy != null )
\t\t\t\t{
\t\t\t\t\tItem[] Content = toCopy.FindItemsByType( typeof( Item ) );
\t\t\t\t\tcopy.TotalGold = 0;\t\t\t\t\t
\t\t\t\t\tcopy.TotalItems = 0;
\t\t\t\t\tcopy.TotalWeight = 0;
\t\t\t\t\tfor ( int i = 0; i < Content.Length; ++i)
\t\t\t\t\t{
\t\t\t\t\t\tif ( Content[i].Parent == toCopy )
\t\t\t\t\t\t{
\t\t\t\t\t\t\tif ( Content[i] is Container )
\t\t\t\t\t\t\t\tcopyContainer( from, copy, (Container)Content[i] );
\t\t\t\t\t\t\telse
\t\t\t\t\t\t\t\tcopyItem( from, copy, Content[i] );
\t\t\t\t\t\t}
\t\t\t\t\t}
\t\t\t\t}
\t\t\t}
\t\t}
\t}
}
