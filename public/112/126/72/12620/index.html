Changement de layer

Accès : Joueurs
Commande : .ChangeLayer
Effet : Change le layer d’un objet* parmis un groupe**.
Usage : Tapez la commande, puis sélectionnez l’objet*.

  • Conditions de l’objet décrites dans le fonctionnement.
    ** Les différents groupes sont définis dans le fonctionnement.

Fonctionnement :

Une fois l’objet sélectionné, le script fait les vérifications suivantes :

  • L’objet est un vêtement ( BaseClothing ), le cas contraire il renvoie un message d’erreur à l’utilisateur.

« Vous ne pouvez pas changer le layer de cela. »

  • L’objet se trouve dans le sac de l’utilisateur de la commande, sinon il renvoie un message d’erreur à l’utilisateur.

« Cela doit être dans votre sac pour changer le layer. »

  • Le layer de l’objet, si c’est un vêtement, appartient à un des trois groupes ci-dessous, autrement il renvoie un message d’erreur à l’utilisateur.

« Vous ne pouvez pas changer le layer de ce vêtement. »

Les trois groupes sont les suivants :

Le groupe dit « tête ».

  • Helm : Tout ce qui est chapeau, casque, turban, ce layer passe au-dessus des cheveux.
  • Neck : Ce qui se rapporte au cou, ce layer passe en dessous des cheveux.
  • Earrings : Tout ce qui est boucle d’oreilles ( et actuellement en jeu les foulards sauf le col ), ce layer passe au dessus des cheveux.

Le groupe dit « jambes ».

  • Pants : Le pantalon, niveau le plus bas sur les jambes ( sauf erreur de ma part ).
  • OuterLegs : Les jupes qui vont par-dessus le pantalon.

A noter que le niveau InnerLegs n’a pas été repris car il semble inutilisable. Le cas contraire, la structure du script permet de facilement le rajouter.

Le groupe dit « torse ».

  • Shirt : Tout ce qui est chemise, donc proche du corps, normalement le niveau le plus bas.
  • InnerTorso : Niveau juste au dessus de la chemise.
  • MiddleTorso : Par pure logique, le niveau au dessus de InnerTorso.
  • OuterTorso : Le niveau le plus haut sur le torse. A noter que ce dernier cache certains niveaux inférieurs !

Une fois validé, le script vérifie que le vêtement n’a pas été supprimé.
Il vérifie également que le vêtement est toujours dans votre sac.

« Le vêtement doit rester dans votre sac ! »

Ensuite il détermine simplement si vous avez changé le layer ou non.

« Vous changez le Layer en : »
« Vous gardez le même Layer. »\t

Si on ne valide rien dans le gump…

« Vous décidez de ne pas changer le Layer. »

Note(s) supplémentaire(s) :

  • Le script est aisément modifiable et modulable pour la création de nouveaux groupes et ou le changement des conditions.

\t\tpublic static void Initialize()
\t\t{

\t\t}

\t\t[Description( "Change le layer d'un objet sous conditions." )]
\t\tprivate static void ChangeLayer_OnCommand( CommandEventArgs e )
\t\t{

\t\t\te.Mobile.SendMessage( "De quel vêtement voulez-vous changer le layer ?" );
\t\t}
\t\t
\t\tprivate class ChangeLayerTarget : Target
\t\t{
\t\t\tpublic ChangeLayerTarget() : base( -1, false, TargetFlags.None )
\t\t\t{
\t\t\t}
\t\t\t
\t\t\tprotected override void OnTarget( Mobile from, object target )
\t\t\t{
\t\t\t\t// Vérifie que la cible est bien un vêtement ( non une armure ou quoique ce soit d'autre ).
\t\t\t\t
\t\t\t\tif ( !(target is BaseClothing) || target is BaseHat )
\t\t\t\t{
\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas changer le layer de cela." );
\t\t\t\t\treturn;
\t\t\t\t}
\t\t\t\t
\t\t\t\tBaseClothing targ = (BaseClothing) target;
\t\t\t\t
\t\t\t\t// Vérifie que l'objet est dans le sac.
\t\t\t\t
\t\t\t\tif ( !targ.IsChildOf( from.Backpack ) )
\t\t\t\t{
\t\t\t\t\tfrom.SendMessage( "Cela doit être dans votre sac pour changer le layer." );
\t\t\t\t\treturn;
\t\t\t\t}
\t\t\t\t
\t\t\t\tif ( targ.Layer == Layer.Helm || targ.Layer == Layer.Neck || targ.Layer == Layer.Earrings ||
\t\t\t\t targ.Layer == Layer.Pants || targ.Layer == Layer.OuterLegs || /targ.Layer == Layer.InnerLegs ||/
\t\t\t\t targ.Layer == Layer.Shirt || targ.Layer == Layer.OuterTorso || targ.Layer == Layer.MiddleTorso || targ.Layer == Layer.InnerTorso )
\t\t\t\t{
\t\t\t\t\tfrom.CloseGump( typeof( ChangeLayerGump ) );
\t\t\t\t\tfrom.SendGump( new ChangeLayerGump( targ ) );
\t\t\t\t}
\t\t\t\t
\t\t\t\telse
\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas changer le layer de ce vêtement." );
\t\t\t}
\t\t\t

\t\t\t{
\t\t\t\tprivate BaseClothing m_Cloth;
\t\t\t\tprivate ChangeLayerEntry[] m_Entry;

\t\t\t\t{
\t\t\t\t\tprivate Layer m_Layer;

\t\t\t\t\tpublic Layer getLayer
\t\t\t\t\t{
\t\t\t\t\t\tget
\t\t\t\t\t\t{
\t\t\t\t\t\t\treturn m_Layer;
\t\t\t\t\t\t}
\t\t\t\t\t}

\t\t\t\t\tpublic ChangeLayerEntry( Layer layer )
\t\t\t\t\t{
\t\t\t\t\t\tm_Layer = layer;
\t\t\t\t\t}
\t\t\t\t}

\t\t\t\tprivate static ChangeLayerEntry[] m_Head = new ChangeLayerEntry[]
\t\t\t\t{
\t\t\t\t\tnew ChangeLayerEntry( Layer.Neck ),
\t\t\t\t\tnew ChangeLayerEntry( Layer.Helm ),
\t\t\t\t\tnew ChangeLayerEntry( Layer.Earrings )
\t\t\t\t};
\t\t\t\t
\t\t\t\tprivate static ChangeLayerEntry[] m_Legs = new ChangeLayerEntry[]
\t\t\t\t{
\t\t\t\t\tnew ChangeLayerEntry( Layer.Pants ),
\t\t\t\t\tnew ChangeLayerEntry( Layer.OuterLegs )/,
\t\t\t\t\tnew ChangeLayerEntry( Layer.InnerLegs )/
\t\t\t\t};
\t\t\t\t
\t\t\t\tprivate static ChangeLayerEntry[] m_Torso = new ChangeLayerEntry[]
\t\t\t\t{
\t\t\t\t\tnew ChangeLayerEntry( Layer.Shirt ),
\t\t\t\t\tnew ChangeLayerEntry( Layer.OuterTorso ),
\t\t\t\t\tnew ChangeLayerEntry( Layer.MiddleTorso ),
\t\t\t\t\tnew ChangeLayerEntry( Layer.InnerTorso )

\t\t\t\tpublic ChangeLayerGump( BaseClothing cloth ) : base( 50, 50 )
\t\t\t\t{
\t\t\t\t\tm_Cloth = cloth;\t\t\t\t\t
\t\t\t\t\t
\t\t\t\t\tif ( m_Cloth.Layer == Layer.Helm || m_Cloth.Layer == Layer.Neck || m_Cloth.Layer == Layer.Earrings )
\t\t\t\t\t\tm_Entry = m_Head;
\t\t\t\t\telse if ( m_Cloth.Layer == Layer.Pants || m_Cloth.Layer == Layer.OuterLegs /|| m_Cloth.Layer == Layer.InnerLegs/ )
\t\t\t\t\t\tm_Entry = m_Legs;
\t\t\t\t\telse
\t\t\t\t\t\tm_Entry = m_Torso;

\t\t\t\t\tAddPage( 0 );

\t\t\t\t\tAddBackground( 100, 10, 350, 96 + m_Entry.Length * 22, 2600 );

\t\t\t\t\t

\t\t\t\t\tAddButton( 149, 70 + m_Entry.Length * 22, 4005, 4007, 1, GumpButtonType.Reply, 0 );
\t\t\t\t\tAddLabel( 185, 71 + m_Entry.Length * 22, 0x00, "Changer le layer ! " );
\t\t\t\t\t
\t\t\t\t\tfor ( int i = 0; i < m_Entry.Length; ++i )
\t\t\t\t\t{
\t\t\t\t\t\tAddLabel( 130, 59 + i * 22, 0x00, "" + m_Entry[i].getLayer );
\t\t\t\t\t\tAddRadio( 207, 60 + i * 22, 210, 211, false, i );
\t\t\t\t\t}
\t\t\t\t}

\t\t\t\tpublic override void OnResponse( NetState from, RelayInfo info )
\t\t\t\t{
\t\t\t\t\tif ( m_Cloth.Deleted )
\t\t\t\t\t\treturn;

\t\t\t\t\tMobile m = from.Mobile;
\t\t\t\t\tint[] switches = info.Switches;

\t\t\t\t\t// Vérifie que le vêtement est toujours dans le sac
\t\t\t\t\tif ( !m_Cloth.IsChildOf( m.Backpack ) )
\t\t\t\t\t{
\t\t\t\t\t\tm.SendMessage( "Le vêtement doit rester dans votre sac !" );
\t\t\t\t\t\treturn;
\t\t\t\t\t}

\t\t\t\t\tif ( info.ButtonID != 0 && switches.Length > 0 )
\t\t\t\t\t{
\t\t\t\t\t\tif ( switches[0] < m_Entry.Length )
\t\t\t\t\t\t{
\t\t\t\t\t\t\tChangeLayerEntry e = m_Entry[switches[0]];
\t\t\t\t\t\t\tif ( m_Cloth.Layer != e.getLayer )
\t\t\t\t\t\t\t{
\t\t\t\t\t\t\t\tm_Cloth.Layer = e.getLayer;
\t\t\t\t\t\t\t\tm.SendMessage( "Vous changez le Layer en : " + e.getLayer );
\t\t\t\t\t\t\t}
\t\t\t\t\t\t\telse
\t\t\t\t\t\t\t\tm.SendMessage( "Vous gardez le même Layer." );
\t\t\t\t\t\t}
\t\t\t\t\t}
\t\t\t\t\telse
\t\t\t\t\t{
\t\t\t\t\t\tm.SendMessage( "Vous décidez de ne pas changer le Layer." );
\t\t\t\t\t}
\t\t\t\t}
\t\t\t}
\t\t}
\t}
}
