Je laisse le script ici pour en avoir une trace...

Le script est compatible normallement avec toutes les versions de RunUo, crépuscule ou non

Pour le gump, j'ai repris celui que Maharet m'a conseillé, sauf modif mineur.

Pour les déplacements :
Haut/Bas : Check si l'objet peut être monter ou descendre
Flip : Ajouté sur le boutton du milieu !
Direction : Check si y'a un mur ou non, donc impossible de faire passer des items à travers les murs

Donc, voici le monstre :

\t\t[Usage( "DecoMove" )]
\t\t[Description( "Déplace un objet adjacent sous conditions." )]
\t\tprivate static void DecoMove_OnCommand( CommandEventArgs e )
\t\t{
\t\t\te.Mobile.Target = new DecoMoveTarget();
\t\t\te.Mobile.SendMessage( "Quel objet voulez-vous déplacer ?" );
\t\t}

\t\tprivate class DecoMoveTarget : Target
\t\t{
\t\t\tpublic DecoMoveTarget() : base( -1, false, TargetFlags.None )
\t\t\t{
\t\t\t}

\t\t\tprotected override void OnTarget( Mobile from, object target )
\t\t\t{
\t\t\t\tif ( !(target is Item) )
\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas déplacer ceci." );
\t\t\t\telse
\t\t\t\t{
\t\t\t\t\tItem targ = (Item) target;

\t\t\t\t\tif ( targ.TotalWeight + targ.PileWeight > 100 )
\t\t\t\t\t\tfrom.SendMessage( "C'est trop lourd pour que vous puissiez le déplacer." );

\t\t\t\t\telse if ( !from.InRange( targ.GetWorldLocation(), 1 ) )
\t\t\t\t\t\tfrom.SendMessage( "C'est hors de portée." );
\t
\t\t\t\t\telse if ( !targ.Movable && !targ.IsLockedDown )
\t\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas bouger ceci." );
\t\t\t\t\telse
\t\t\t\t\t{
\t\t\t\t\t\tfrom.CloseGump( typeof( DecoMoveGump ) );
\t\t\t\t\t\tfrom.SendGump( new DecoMoveGump( targ ) );
\t\t\t\t\t}
\t\t\t\t}
\t\t\t}

\t\t\t\tpublic bool checkMoveAccess( Mobile from, Item targ )
\t\t\t\t{
\t\t\t\t\tif ( targ.TotalWeight + targ.PileWeight > 100 )
\t\t\t\t\t\tfrom.SendMessage( "C'est trop lourd pour que vous puissiez le déplacer." );
\t\t
\t\t\t\t\telse if ( !from.InRange( targ.GetWorldLocation(), 1 ) )
\t\t\t\t\t\tfrom.SendMessage( "C'est hors de portée." );
\t\t
\t\t\t\t\telse if ( !targ.Movable && !targ.IsLockedDown )
\t\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas bouger ceci." );
\t\t\t\t\telse
\t\t\t\t\t\treturn true;
\t
\t\t\t\t\treturn false;
\t\t\t\t}

\t\t\t\tprivate int GetFloorZ( Item item )
\t\t\t\t{
\t\t\t\t\tMap map = item.Map;
\t\t
\t\t\t\t\tif ( map == null )
\t\t\t\t\t\treturn int.MinValue;
\t\t
\t\t\t\t\tTile[] tiles = map.Tiles.GetStaticTiles( item.X, item.Y, true );

\t\t\t\t\tint z = int.MinValue;
\t
\t\t\t\t\tfor ( int i = 0; i < tiles.Length; ++i )
\t\t\t\t\t{
\t\t\t\t\t\tTile tile = tiles[i];
\t\t\t\t\t\tItemData id = TileData.ItemTable[tile.ID & 0x3FFF];
\t\t\t\t
\t\t\t\t\t\tint top = tile.Z; // Confirmed : no height checks here
\t\t
\t\t\t\t\t\tif ( id.Surface && !id.Impassable && top > z && top <= item.Z )
\t\t\t\t\t\t\tz = top;
\t\t\t\t\t}
\t\t\t\t\treturn z;
\t\t\t\t}

\t\t\t\tprivate bool Up( Item item, Mobile from )
\t\t\t\t{
\t\t\t\t\tint floorZ = GetFloorZ( item );
\t\t
\t\t\t\t\tif ( floorZ > int.MinValue && item.Z < (floorZ + 15) )
\t\t\t\t\t{
\t\t\t\t\t\titem.Location = new Point3D( item.Location, item.Z + 1 );
\t\t\t\t\t\treturn true;
\t\t\t\t\t}
\t\t\t\t\telse
\t\t\t\t\t{
\t\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas monter cet objet plus haut." );
\t\t\t\t\t\treturn false;
\t\t\t\t\t}
\t\t\t\t}

\t\t\t\tprivate bool Down( Item item, Mobile from )
\t\t\t\t{
\t\t\t\t\tint floorZ = GetFloorZ( item );
\t\t
\t\t\t\t\tif ( floorZ > int.MinValue && item.Z > GetFloorZ( item ) )
\t\t\t\t\t{
\t\t\t\t\t\titem.Location = new Point3D( item.Location, item.Z - 1 );
\t\t\t\t\t\treturn true;
\t\t\t\t\t}
\t\t\t\t\telse
\t\t\t\t\t{
\t\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas descendre cet objet plus bas." );
\t\t\t\t\t\treturn false;
\t\t\t\t\t}
\t\t\t\t}

\t\t\t\tprivate static void Turn( Item item, Mobile from )
\t\t\t\t{
\t\t\t\t\tFlipableAttribute[] attributes = (FlipableAttribute[])item.GetType().GetCustomAttributes( typeof( FlipableAttribute ), false );
\t\t
\t\t\t\t\tif( attributes.Length > 0 )
\t\t\t\t\t\tattributes[0].Flip( item );
\t\t\t\t\telse
\t\t\t\t\t\tfrom.SendMessage( "Vous ne pouvez pas tourner cet objet." );
\t\t\t\t}
